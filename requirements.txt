  <h1>Toronto Drop‑In Sports Finder</h1>
  <p>Find drop‑in badminton and pickleball sessions at City of Toronto recreation facilities.</p>
  <form id="searchForm">
    <label for="startDate">Start date:</label>
    <input type="date" id="startDate" name="startDate" required>

    <label for="endDate">End date:</label>
    <input type="date" id="endDate" name="endDate" required>

    <label for="sport">Sport:</label>
    <select id="sport" name="sport">
      <option value="both">Badminton or Pickleball</option>
      <option value="badminton">Badminton only</option>
      <option value="pickleball">Pickleball only</option>
    </select>

    <button type="submit">Search</button>
  </form>
  <div id="message"></div>
  <div id="results"></div>

  <script>
    const form = document.getElementById("searchForm");
    const messageEl = document.getElementById("message");
    const resultsEl = document.getElementById("results");

    form.addEventListener("submit", async function(e) {
      e.preventDefault();
      // Clear previous state
      messageEl.textContent = "";
      resultsEl.innerHTML = "";
      const startDate = document.getElementById("startDate").value;
      const endDate = document.getElementById("endDate").value;
      const sport = document.getElementById("sport").value;
      if (!startDate || !endDate) {
        messageEl.textContent = "Please select both start and end dates.";
        return;
      }
      // Build query string
      const params = new URLSearchParams({
        start_date: startDate,
        end_date: endDate,
        sport: sport
      });
      try {
        const resp = await fetch("/api/dropin?" + params.toString());
        const body = await resp.json();
        if (!resp.ok) {
          messageEl.textContent = body.error || "An error occurred.";
          return;
        }
        const results = body.results || [];
        if (results.length === 0) {
          messageEl.textContent = "No matching drop‑in sessions found.";
          return;
        }
        // Render results
        results.forEach(function(fac) {
          const div = document.createElement("div");
          div.className = "facility";
          const h2 = document.createElement("h2");
          h2.textContent = fac.facility;
          div.appendChild(h2);
          const ul = document.createElement("ul");
          fac.events.forEach(function(ev) {
            const li = document.createElement("li");
            li.textContent = ev.date + " at " + ev.time + " | Age: " + ev.age + " | Sport: " + ev.sport;
            ul.appendChild(li);
          });
          div.appendChild(ul);
          resultsEl.appendChild(div);
        });
      } catch(err) {
        messageEl.textContent = err.message || "Failed to fetch results.";
      }
    });
  </script>
